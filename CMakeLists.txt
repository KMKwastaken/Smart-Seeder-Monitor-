cmake_minimum_required(VERSION 3.16)
project(SmartSeeder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(seeder_sim simulator/main.cpp)
if(MSVC)
 target_compile_options(seeder_sim PRIVATE /W4)
else()
 target_compile_options(seeder_sim PRIVATE -Wall -Wextra -Wpedantic)
endif()
